/// Hash rápido de datos - CORREGIDO sin overflow
/// Usa wrapping arithmetic para evitar panic
pub export fn fast_hash(
    data_ptr: [*]const u8,
    data_len: usize,
) u64 {
    const data = data_ptr[0..data_len];
    var hash: u64 = 5381;
    
    for (data) |byte| {
        // ✅ CORREGIDO: Usar wrapping operators
        const shifted = hash *% 32;  // Wrapping shift-left
        hash = shifted +% hash +% @as(u64, byte);  // Wrapping add
    }
    
    return hash;
}
